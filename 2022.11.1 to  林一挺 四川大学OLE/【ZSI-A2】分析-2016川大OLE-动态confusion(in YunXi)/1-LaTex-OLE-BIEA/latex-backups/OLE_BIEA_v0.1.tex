\begin{abstract}

Recently, a bit-level image encryption algorithm based on chaotic maps was proposed \cite{LuX2016OLE}, in which a diffusion-permutation structure with only one round is adopted. In this paper, we performed a comprehensive cryptanalysis on the BCIEA. Theoretical analysis and experimental simulations verify that BCIEA cannot resist a chosen ciphertext attack method.


\end{abstract}

\subsection{Description of BCIEA}

The block diagram of BCIEA is shown in Fig. . As can be seen, the main components of BCIEA are introduced briefly as follows:

\begin{itemize}
	
	\item \emph{The chaotic map and the Secret Key}:
	
	In BCIEA, a chaotic map is used twice to generate several keystreams for encryption. The piece-wise linear chaotic map (PWLCM) is mathematically modeled by
	\begin{equation}
	\begin{aligned}
	x_i = F(x_{i-1},\eta)=	\begin{cases}
	x_{i-1}/ \eta,              & 0<x_{i-1}<\eta \\
	(x_{i-1}- \eta)/(0.5-\eta), & \eta \le x_{i-1}<0.5 \\
	F(1-x_{i-1},\eta),          & 0.5 \le x_{i-1}<1 \\
	\end{cases}
	\end{aligned}
	\label{eq:PWLCM}
	\end{equation}
	where $x$ is the state variable, $x_0 \in (0,1)$ is the initial value and $\eta \in (0,0.5)$ is the control parameter, respectively. In BCIEA, two different groups of  initial values and control parameters $(x_0,\eta_1,y_0,\eta_2)$ are served as the secret keys.
	
	\item \emph{Initialization}:
	For the sake of generality, the encrypted object is an 8-bit grayscale image of size $\bm{P}$ of size $M \times N$ (height $\times$ width). The initialization operation consists of two parts: one is converting the plain image into its bit-level form, and the other is generating the keystreams by the chaotic map for the first time.
	\begin{itemize}
		\item By BBD, convert the plain image $\bm{P}$ of size $M \times N$ (height $\times$ width) into two 1D sequences $\bm{A}_1$ and $\bm{A}_2$ of length $4MN$. For convenience, let $L$ be $4MN$.
		\item With the initial parameter $\eta_1$ and the initial value $x_0$, get a sequence $\bm{X}_1$ by iterating Eq. (\ref{eq:PWLCM}) for $N_0+MN$ times and discard the former $N_0$ values, given as
		\begin{equation*}
		X_1(i) = \bmod(floor(x(i))\times 10^{14},256)\\
		\label{eq:PRNS}
		\end{equation*}
		where $i =1 \sim MN$, $floor$ is the function that takes an integer, and $X_1(i) \in \mathbb{Z}_{256}$. Then, two sequences $\bm{b}_1,\bm{b}_2$ of length $L$ are obtained from $\bm{X_1}$ by the method of binary bitplane decomposition. Here, $\bm{b}_1,\bm{b}_2 \in \mathbb{Z}_{256}$.
	\end{itemize}
	
	
	\item \emph{Stage 1. diffusion}:
	
	Step 1. Firstly calculate the sum of $\bm{A}_2$, represented as
	\begin{equation}
	sum_1 = \sum\limits_{i= 1}^{L} A_2(i)
	\label{eq:sum1}
	\end{equation}
	where $L=4MN$. Then run a cyclic right shift operation on $\bm{A}_1$ with $sum_1$ bits to obtain $\bm{A}_{11}$.
	
	Step. 2 Get $\bm{B}_{1}$ by performing the bitwise XOR operations between $\bm{A}_{11}$, $\bm{A}_{2}$ and $\bm{b}_{1}$, given as
	\begin{equation}
	\left\{ \begin{array}{l}
	B_1(1) = A_{11}(1) \oplus A_{11}(L) \oplus A_{2}(1) \oplus b_{1}(1)\\
	B_1(i) = A_{11}(i) \oplus A_{11}(i-1) \oplus A_{2}(i) \oplus b_{1}(i)
	\end{array} \right.
	\label{eq:B1_diffusion}
	\end{equation}
	where $i=2 \sim L$.
	
	Step 3. Like Step 1, firstly compute the sum values $sum_2$ of $\bm{B}_1$ as
	\begin{equation}
	sum_2 = \sum\limits_{i= 1}^{L} B_1(i)
	\label{eq:sum2}
	\end{equation}
	Then , get $\bm{A}_{22}$ cyclic right shifting $\bm{A}_2$ with $sum_2$ bits.
	
	Step. 4 Similar to Step 2, obtain $\bm{B}_{2}$ from $\bm{A}_{22}$, $\bm{B}_{1}$ and $\bm{b}_{2}$, modeled by
	\begin{equation}
	\left\{ \begin{array}{l}
	B_2(1) = A_{22}(1) \oplus A_{22}(L) \oplus B_{1}(1) \oplus b_{2}(1)\\
	B_2(i) = A_{22}(i) \oplus A_{22}(i-1) \oplus B_{1}(i) \oplus b_{2}(i)
	\end{array} \right.
	\label{eq:B2_diffusion}
	\end{equation}
	where $i=2 \sim L$.
	
	\item \emph{Stage 2. Confusion}:
	
	Step 1. Firstly compute the sum values of $\bm{B}_1$ and $\bm{B}_1$ by
	\begin{equation}
	sum = \sum\limits_{i= 1}^{L} (B_1(i)+B_2(i))
	\label{eq:sum3}
	\end{equation}
	Further update the initial value of the chaotic map by
	\begin{equation}
	s_0 = \bmod(y_0+sum/L,1)\\
	\label{eq:update_y0}
	\end{equation}
	Then, with the initial parameter $\eta_2$ and the initial value $s_0$,  using Eq. (\ref{eq:PWLCM}) again, generate a chaotic sequences of length $2L$, and then process it into two integer sequences $\bm{Y}$ and $\bm{Z}$ both of length $L$. Here, $\bm{Y},\bm{Z} \in [1,L]$.
	
	Step 2. Use $\bm{Y}$ and $\bm{Z}$ as coordinate indexes, and then get $\bm{C}_1$ and $\bm{C}_2$ by confusing the positions of the elements in $\bm{B}_2$ and $\bm{B}_1$ respectively. Exactly, given as
	\begin{equation}
	\left\{ \begin{array}{l}
	C_1(i) = B_2(Y(i))\\
	C_2(i) = B_1(Z(i))\\
	\end{array} \right.
	\label{eq:confusion}
	\end{equation}
	where $i=1 \sim L$. 
	
	Finally, get the cipher image $\bm{C}$ from $\bm{C}_1$ and $\bm{C}_2$ by the method of BPC.
\end{itemize}
As stated in \cite{LuX2016OLE}, the decryption process is the inverse of the encryption process, but the specific steps are not described. For more details, please refer to \cite{LuX2016OLE}. 

\section{Cryptanalysis}
\label{sec3:Cryptanalysis}

\subsection{On the decryption of BCIEA}

As everyone knows, ensuring the decryption is available is a basic condition for good cryptosystems. In addition, decryption is an integral part of cryptanalysis. For this reason, we discuss the decryption part firstly.

\begin{itemize}
	
	\item \emph{Initialization}:
	As with encryption, two initialization operations are performed:
	\begin{itemize}
		\item Get $\bm{C}_1$ and $\bm{C}_2$ from the cipher image $\bm{C}$ by BPD. 
		
		\item Generate the keystreams $\bm{b}_1,\bm{b}_2$ by the secret keys $(\eta_1,x_0)$.
	\end{itemize}

	
	
	\item \emph{Stage 1. Anti-confusion}:
	
	Step 1. Firstly compute the sum values of $\bm{C}_1$ and $\bm{C}_2$ by
	\begin{equation}
	sum = \sum\limits_{i= 1}^{L} (C_1(i)+C_2(i))
	\label{eq:sum_C}
	\end{equation}
	Note that the sum of $\bm{B}_1$ and $\bm{B}_2$ is equal to the sum of $\bm{C}_1$ and $\bm{C}_2$ because the confusion only changes the positions of the elements.	Then, with the same secret keys $(\eta_2,y_0)$, update the initial value $y_0$ to $s_0$ and further obtain the two coordinate indexes sequences $\bm{Y}$ and $\bm{Z}$.
	
	Step 2. Restore $\bm{B}_1$ and $\bm{B}_2$ from $\bm{C}_2$ and $\bm{C}_1$ by the anti-confuse, given by
	\begin{equation}
	\left\{ \begin{array}{l}
	B_1(Z(i)) = C_2(i)\\
    B_2(Y(i)) = C_1(i)
	\end{array} \right.
	\label{eq:anti-confusion}
	\end{equation}
	where $i=1 \sim L$. 
	
	
	\item \emph{Stage 2. Anti-diffusion}:
	
	Step 1. Corresponding to Step 4 of the confusion, recover $\bm{A}_{22}$ from $\bm{B}_{2}$, $\bm{B}_{1}$ and $\bm{b}_{2}$, modeled by
	\begin{equation}
	\left\{ \begin{array}{l}
	A_{22}(1) \oplus A_{22}(L)  = B_2(1) \oplus B_{1}(1) \oplus b_{2}(1)\\
	A_{22}(i) \oplus A_{22}(i-1) = B_2(i) \oplus B_{1}(i) \oplus b_{2}(i)
	\end{array} \right.
	\label{eq:B2_to_A22}
	\end{equation}
	where $i=2 \sim L$.
	
	Step 2. Obtain $sum_2$ by Eq. (\ref{eq:sum2}), and then get $\bm{A}_{2}$ by doing cyclic left shift operation on $\bm{A}_22$ with $sum_2$ bits.
	
%	Step 1. Firstly calculate the sum of $\bm{A}_2$, represented as
%	\begin{equation}
%	sum_1 = \sum\limits_{i= 1}^{L} A_2(i)
%	\label{eq:sum1}
%	\end{equation}
%	where $L=4MN$. Then run a cyclic right shift operation on $\bm{A}_1$ with $sum_1$ bits to obtain $\bm{A}_{11}$.
	
	Step. 4 Recover $\bm{A}_{11}$ by performing the bitwise XOR operations between $\bm{B}_{1}$, $\bm{A}_{2}$ and $\bm{b}_{1}$, given as
	\begin{equation}
	\left\{ \begin{array}{l}
	A_{11}(1) \oplus A_{11}(L) = B_1(1) \oplus A_{2}(1) \oplus b_{1}(1)\\
	A_{11}(i) \oplus A_{11}(i-1) B_1(i) \oplus A_{2}(i) \oplus b_{1}(i)
	\end{array} \right.
	\label{eq:B1_to_A11}
	\end{equation}
	where $i=2 \sim L$.
	
	Step 4. Obtain $sum_1$ by Eq. (\ref{eq:sum1}), and then get $\bm{A}_{1}$ by doing cyclic left shift operation on $\bm{A}_11$ with $sum_1$ bits.
	
	Finally, get the cipher image $\bm{C}$ from $\bm{C}_1$ and $\bm{C}_2$ by the method of BPC.
\end{itemize}

\section{Conclusion}


This paper performed a comprehensive cryptanalysis on a chaotic image encryption scheme based on a variant of Hill cipher so-called BCIEA. We found that BCIEA cannot resist a chosen plaintext attack because of its inherent security defects. Both theoretical analysis and experimental verification were provided to support that BCIEA can be broken with low computational and data complexity. This study can provide a reference for improving the security of image encryption technology.
